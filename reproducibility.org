def load_msa(msa_file, alphabet=None, pad_char='-'):
    """Load an MSA file and return encoded tensor X, alphabet, and length."""
    seqs = list(read_fasta(msa_file, seq_type="nuc").values())

    if alphabet is None:
        chars = sorted(set("".join(seqs)))
        if pad_char not in chars:
            chars.append(pad_char)
        alphabet = chars

    L = max(len(s) for s in seqs)
    stoi = {c: i for i, c in enumerate(alphabet)}

    X = torch.tensor(
        [[stoi.get(c, stoi[pad_char]) for c in s.ljust(L, pad_char)] for s in seqs],
        dtype=torch.long,
    )
    return X, alphabet, L


class RNADataset(Dataset):
    def __init__(self, X):
        self.X = X

    def __len__(self):
        return self.X.size(0)

    def __getitem__(self, i):
        return self.X[i]


def train_vae_on_msa(msa_file, epochs=100, batch_size=32, lr=1e-3, weight_decay=1e-5, beta=1.0, emb=5, latent_dim=16, hidden=128, alphabet=None, save_dir=".",):
    X, alphabet, L = load_msa(msa_file, alphabet=alphabet)
    dl = DataLoader(RNADataset(X), batch_size=batch_size, shuffle=True)

    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model = VAE(
        seq_len=L,
        nb_state=len(alphabet),
        emb=emb,
        latent_dim=latent_dim,
        hidden=hidden,
    ).to(device)

    opt = optim.Adam(model.parameters(), lr=lr, weight_decay=weight_decay)

    for ep in range(1, epochs + 1):
        model.train()
        tot = ce = kl = n = 0.0
        for xb in dl:
            xb = xb.to(device)
            opt.zero_grad()
            out = model(xb)
            losses = model.loss(xb, out, beta=beta)
            losses["total"].backward()
            nn.utils.clip_grad_norm_(model.parameters(), 1.0)
            opt.step()
            b = xb.size(0)
            tot += losses["total"].item() * b
            ce += losses["ce"].item() * b
            kl += losses["kl"].item() * b
            n += b

    save_path = Path(save_dir) / f"vae_{Path(msa_file).stem}.pt"
    torch.save(
        {
            "model_state": model.state_dict(),
            "seq_len": model.seq_len,
            "nb_state": model.nb_state,
            "emb": model.emb,
            "alphabet": alphabet,
        },
        save_path,
    )
    return model, save_path

#!/usr/bin/env python3
# pip install logomaker
import os
import re
from glob import glob

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import logomaker

from src.utils import read_fasta_no_insertions, msa_to_oh

# IMPORTANT: set this to the SAME column order msa_to_oh uses.
# The matrix plot is robust to this; only the logo's letter labels depend on it.
ALPHABET = list("ACDEFGHIKLMNPQRSTVWY-")


def load_oh(infile):
    msa = read_fasta_no_insertions(infile, seq_type="protein")
    X = msa_to_oh(list(msa.values()), seq_type="protein")   # (N, L, k)
    return X


def view_msa(infile, max_rows=500):
    X = load_oh(infile)
    N, L, k = X.shape
    alphabet = ALPHABET[:k]

    fig, (ax0, ax1) = plt.subplots(
        2, 1, figsize=(min(0.05 * L + 4, 20), 8),
        gridspec_kw={"height_ratios": [3, 1]},
    )

    # --- 1) MSA as a colored matrix (one color per residue identity) ---
    M = X.argmax(-1)                                  # (N, L) integer-encoded
    if N > max_rows:                                  # subsample huge MSAs to display
        M = M[np.random.choice(N, max_rows, replace=False)]
    im = ax0.imshow(M, aspect="auto", interpolation="nearest",
                    cmap="tab20b", vmin=0, vmax=k - 1)
    ax0.set_ylabel("sequence")
    ax0.set_title(os.path.basename(infile))
    cbar = fig.colorbar(im, ax=ax0, ticks=range(k))
    cbar.ax.set_yticklabels(alphabet)

    # --- 2) sequence logo (information content, bits) ---
    prob = pd.DataFrame(X.mean(0), columns=alphabet)  # (L, k) position frequencies
    if "-" in prob.columns:                           # logos usually exclude gaps
        prob = prob.drop(columns="-")
        prob = prob.div(prob.sum(1), axis=0).fillna(0)
    info = logomaker.transform_matrix(prob, from_type="probability",
                                      to_type="information")
    logomaker.Logo(info, ax=ax1, color_scheme="chemistry")
    ax1.set_xlabel("position")
    ax1.set_ylabel("bits")

    fig.tight_layout()
    return fig
